### PASSO 1: CRIAR O MODELO `Noticia.cs` ###

# O modelo é a representação da nossa tabela em C#.

# 1. Crie o arquivo: `Api/webApi/Models/Noticia.cs`
# 2. Adicione o seguinte código:

```csharp
// ---- CONTEÚDO PARA Api/webApi/Models/Noticia.cs ----

using webApi.Models.Base;
using System; // Necessário para o DateTime

namespace webApi.Models
{
    // A classe Noticia herda de ModelBase para já ter Id, CreatedAt, etc.
    public class Noticia : ModelBase
    {
        public string Title { get; set; }
        public string Content { get; set; }
        public DateTime PublishedDate { get; set; }

        // Define a chave estrangeira para o relacionamento com a tabela Institution.
        // Isso garante que cada notícia pertence a uma instituição.
        public int InstitutionId { get; set; }
        public Institution Institution { get; set; }
    }
}

PASSO 2: CRIAR O MAPEAMENTO NoticiaMap.cs
O mapeamento define as regras da tabela no banco de dados (nome da tabela,
tipos de coluna, tamanhos, relacionamentos, etc.).
1. Crie o arquivo: Api/webApi/Context/Mappings/NoticiaMap.cs
2. Adicione o seguinte código:
// ---- CONTEÚDO PARA Api/webApi/Context/Mappings/NoticiaMap.cs ----

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using webApi.Models;

namespace webApi.Context.Mappings
{
    public class NoticiaMap : IEntityTypeConfiguration<Noticia>
    {
        public void Configure(EntityTypeBuilder<Noticia> builder)
        {
            // Define o nome da tabela no banco de dados
            builder.ToTable("Noticia");

            // Define a chave primária
            builder.HasKey(n => n.Id);

            // Configura a coluna 'Title'
            builder.Property(n => n.Title)
                .IsRequired() // Campo obrigatório
                .HasMaxLength(200); // Tamanho máximo

            // Configura a coluna 'Content'
            builder.Property(n => n.Content)
                .IsRequired()
                .HasColumnType("TEXT"); // Ideal para textos longos

            // Configura a coluna 'PublishedDate'
            builder.Property(n => n.PublishedDate)
                .IsRequired();

            // Configura o relacionamento: Uma Notícia (HasOne) tem uma Instituição.
            // Uma Instituição (WithMany) pode ter várias notícias.
            builder.HasOne(n => n.Institution)
                .WithMany() // Como não há uma lista de Notícias em Institution, WithMany() é suficiente.
                .HasForeignKey(n => n.InstitutionId)
                .IsRequired();
        }
    }
}

PASSO 6: CRIAR O CONTROLLER NoticiaController.cs
O controller define os endpoints (URLs) que o frontend irá chamar.
1. Crie o arquivo: Api/webApi/Controllers/NoticiaController.cs
2. Adicione o esqueleto do controller com a injeção de dependência:
// ---- CONTEÚDO PARA Api/webApi/Controllers/NoticiaController.cs ----

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using webApi.Context;
using webApi.Models;
using System.Threading.Tasks;
using System.Collections.Generic;
using System.Linq;

namespace webApi.Controllers
{
    [ApiController]
    [Route("api/[controller]")] // A rota será /api/noticia
    public class NoticiaController : ControllerBase
    {
        private readonly DataContext _context;

        // O DataContext é injetado automaticamente pelo sistema de injeção de dependência do .NET.
        public NoticiaController(DataContext context)
        {
            _context = context;
        }

        // =======================================================
        // AQUI DENTRO VAMOS INSERIR OS MÉTODOS DO CRUD (C, R, U, D)
        // =======================================================

        // CREATE: Criar uma nova notícia
        // Endpoint: POST /api/noticia
        [HttpPost]
        public async Task<ActionResult<Noticia>> CreateNoticia(Noticia noticia)
        {
            _context.Noticias.Add(noticia);
            await _context.SaveChangesAsync();
            return CreatedAtAction(nameof(GetNoticiaById), new { id = noticia.Id }, noticia);
        }

        // READ (All): Obter todas as notícias
        // Endpoint: GET /api/noticia
        [HttpGet]
        public async Task<ActionResult<IEnumerable<Noticia>>> GetNoticias()
        {
            return await _context.Noticias
                                 .Include(n => n.Institution) // Inclui dados da instituição relacionada
                                 .OrderByDescending(n => n.PublishedDate) // Ordena pelas mais recentes
                                 .ToListAsync();
        }

        // READ (By ID): Obter uma notícia específica
        // Endpoint: GET /api/noticia/5
        [HttpGet("{id}")]
        public async Task<ActionResult<Noticia>> GetNoticiaById(int id)
        {
            var noticia = await _context.Noticias
                                          .Include(n => n.Institution)
                                          .FirstOrDefaultAsync(n => n.Id == id);

            if (noticia == null)
            {
                return NotFound(); // Retorna erro 404 se não encontrar
            }
            return Ok(noticia);
        }

        // UPDATE: Atualizar uma notícia existente
        // Endpoint: PUT /api/noticia/5
        [HttpPut("{id}")]
        public async Task<IActionResult> UpdateNoticia(int id, Noticia noticia)
        {
            if (id != noticia.Id)
            {
                return BadRequest();
            }

            _context.Entry(noticia).State = EntityState.Modified;

            try
            {
                await _context.SaveChangesAsync();
            }
            catch (DbUpdateConcurrencyException)
            {
                if (!_context.Noticias.Any(e => e.Id == id))
                {
                    return NotFound();
                }
                else
                {
                    throw;
                }
            }

            return NoContent(); // Sucesso, sem conteúdo para retornar
        }

        // DELETE: Apagar uma notícia
        // Endpoint: DELETE /api/noticia/5
        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteNoticia(int id)
        {
            var noticia = await _context.Noticias.FindAsync(id);
            if (noticia == null)
            {
                return NotFound();
            }

            _context.Noticias.Remove(noticia);
            await _context.SaveChangesAsync();

            return NoContent();
        }
    }
}
